# Проверка существования ключевых исходных файлов
foreach(file_to_check function.cpp file.cpp metric.cpp metric_accumulator.cpp cmd_options.cpp)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${file_to_check}")

    else()
        message(STATUS "Found ${file_to_check}")
    endif()
endforeach()

# Цель для unit тестов
add_custom_target(unit_tests)

# Функция для удобного добавления библиотек с отладкой
function(add_analyser_library name)
    add_library(${name} ${ARGN})
    
    # Подробные флаги отладки
    target_compile_options(${name} PRIVATE 
        -g3 -ggdb -O0 -fno-omit-frame-pointer
    )
    
    # Директории включения
    target_include_directories(${name} PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_SOURCE_DIR}/include
    )
    
    # Вывод информации о библиотеке
    message(STATUS "Added library: ${name}")
    get_target_property(src_files ${name} SOURCES)
    message(STATUS "  Sources: ${src_files}")
endfunction()

# Библиотека function
add_analyser_library(function
    function.cpp
)

# Библиотека file
add_analyser_library(file
    file.cpp
)

# Библиотека metric
add_analyser_library(metric
    metric.cpp
    metric_impl/code_lines_count.cpp
    metric_impl/cyclomatic_complexity.cpp
    metric_impl/parameters_count.cpp
)

target_link_libraries(metric
    PUBLIC
        function
)

# Библиотека metric_accumulator
add_analyser_library(metric_accumulator
    metric_accumulator.cpp
    metric_accumulator_impl/average_accumulator.cpp
    metric_accumulator_impl/sum_average_accumulator.cpp
)

target_link_libraries(metric_accumulator
    PUBLIC
        function
        file
        metric
)

# Библиотека cmd_options
add_analyser_library(cmd_options
    cmd_options.cpp
)

target_link_libraries(cmd_options

        Boost::program_options
)
